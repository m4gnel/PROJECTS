1. SYSTEM ARCHITECTURE OVERVIEW
1.1 Architecture Principles
Offline-first: Every core feature must function without internet. Internet enhances, not enables.
Provider-independent: No hard dependency on any paid API. Every external service behind an abstraction interface.
Deterministic-first: Rules and cache answer before LLM is called. LLM synthesizes, does not invent.
Evidence-grounded: Every advisory output must carry a traceable evidence bundle — no free-floating AI assertions.
Schema-first: All agent outputs are validated JSON schemas — never free-form text passed between agents.
Cost-aware: Architecture designed for zero infrastructure cost during development and minimum cost during pilot.
Security-by-default: Row-level security, JWT auth, encrypted storage, input validation — from day one.

3.2 Complete Database Schema
Users & Authentication
CREATE TABLE users (
  id            UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  phone         VARCHAR(15) UNIQUE NOT NULL,
  name          VARCHAR(100),
  language      VARCHAR(10) NOT NULL DEFAULT 'ml',  -- BCP-47
  location_lat  DECIMAL(9,6),
  location_lon  DECIMAL(9,6),
  district      VARCHAR(100),
  state         VARCHAR(100) DEFAULT 'Kerala',
  subscription_tier  VARCHAR(20) DEFAULT 'free',
  subscription_expires_at TIMESTAMPTZ,
  trial_started_at  TIMESTAMPTZ DEFAULT NOW(),
  is_active     BOOLEAN DEFAULT true,
  created_at    TIMESTAMPTZ DEFAULT NOW(),
  updated_at    TIMESTAMPTZ DEFAULT NOW(),
  last_active   TIMESTAMPTZ
);
CREATE TABLE otp_sessions (
  id         UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  phone      VARCHAR(15) NOT NULL,
  otp_hash   VARCHAR(255) NOT NULL,  -- bcrypt hashed
  expires_at TIMESTAMPTZ NOT NULL,
  used       BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE refresh_tokens (
  id         UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id    UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  token_hash VARCHAR(255) NOT NULL,
  expires_at TIMESTAMPTZ NOT NULL,
  revoked    BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
Farm & Field Management
CREATE TABLE farms (
  id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id     UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  name        VARCHAR(200) NOT NULL,
  location    GEOMETRY(POINT, 4326),  -- PostGIS point
  district    VARCHAR(100),
  state       VARCHAR(100),
  area_ha     DECIMAL(8,2),
  created_at  TIMESTAMPTZ DEFAULT NOW(),
  updated_at  TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE fields (
  id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  farm_id     UUID NOT NULL REFERENCES farms(id) ON DELETE CASCADE,
  name        VARCHAR(100),
  boundaries  GEOMETRY(POLYGON, 4326),  -- Field boundary
  area_ha     DECIMAL(8,2),
  slope       VARCHAR(20),  -- flat/gentle/moderate/steep
  irrigation  VARCHAR(30),  -- rainfed/canal/drip/sprinkler/tube-well
  soil_type   VARCHAR(50),
  created_at  TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE soil_profiles (
  id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  field_id        UUID NOT NULL REFERENCES fields(id) ON DELETE CASCADE,
  pH              DECIMAL(4,2),
  nitrogen_kgha   DECIMAL(8,2),
  phosphorus_kgha DECIMAL(8,2),
  potassium_kgha  DECIMAL(8,2),
  organic_carbon  DECIMAL(5,2),
  texture         VARCHAR(30),  -- sandy/loamy/clayey/silty
  field_capacity  DECIMAL(5,2),
  wilting_point   DECIMAL(5,2),
  source          VARCHAR(50),  -- soil_health_card/lab_test/estimated
  measured_at     DATE,
  created_at      TIMESTAMPTZ DEFAULT NOW()
);
Crop Cycles & Observations
CREATE TABLE crop_cycles (
  id            UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

  fertilizer_recent   BOOLEAN,
  pesticide_recent    BOOLEAN,
  observed_growth_stage VARCHAR(50),
  observed_at         TIMESTAMPTZ DEFAULT NOW(),
  created_at          TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE images (
  id              UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  observation_id  UUID NOT NULL REFERENCES observations(id) ON DELETE CASCADE,

  mime_type       VARCHAR(50) NOT NULL,
  file_size_bytes INT,
  width_px        INT,
  height_px       INT,
  quality_score   DECIMAL(4,2),  -- 0.0-1.0 from quality checker
  blur_score      DECIMAL(8,2),  -- Laplacian variance
  created_at      TIMESTAMPTZ DEFAULT NOW()
);
Diagnoses & Advisories

Row-Level Security
-- Enable RLS on all user-facing tables
ALTER TABLE farms ENABLE ROW LEVEL SECURITY;
ALTER TABLE fields ENABLE ROW LEVEL SECURITY;
ALTER TABLE crop_cycles ENABLE ROW LEVEL SECURITY;

-- Users see only their own farms
CREATE POLICY user_farms ON farms
  USING (user_id = current_setting('app.current_user_id')::UUID);
-- Cascade via farm ownership for fields, cycles, observations
CREATE POLICY user_fields ON fields
  USING (farm_id IN (SELECT id FROM farms WHERE user_id = current_setting('app.current_user_id')::UUID));

5.3 Critic Agent Logic — Safety Rules
class CriticAgent:
    CONFLICTS = [
        # Vision says fungal but weather is unfavorable
        Rule(vision='fungal', weather_dfi='<0.3', action='REDUCE_CONFIDENCE'),
        # Disease uncommon at current stage
        Rule(vision_label_not_in=stage_disease_map[stage], action='FLAG_STAGE_MISMATCH'),
    ]
    SAFETY_CHECKS = [
        check_chemical_in_cibrc_database,    # BLOCKS if not found
        check_dosage_within_label_range,      # BLOCKS if exceeded
        check_crop_chemical_registration,     # BLOCKS if not registered
        check_preharvest_interval,            # WARNS if PHI not met
        check_reentry_interval,               # WARNS if REI not met
        check_scheme_exists_in_database,      # WARNS if not found
        check_document_not_expired,           # WARNS if valid_until passed
        check_no_llm_invented_products,       # BLOCKS — name must resolve to DB entry
    ]
    def run(self, state: AgriSAGEState) -> CriticResult:
        conflicts = self._detect_conflicts(state)
        safe = all(check(state) for check in self.SAFETY_CHECKS)
        confidence = self._calibrate(state, conflicts)
        missing = self._find_missing_data(state)
        run_sim = confidence.score > 0.65 and not conflicts
        return CriticResult(confidence=confidence, conflicts=conflicts,
                            safe_to_act=safe, missing=missing, run_simulation=run_sim)

6.3 RAG Retrieval Pipeline
# backend/knowledge/retrieval.py
class HybridRetriever:
    def __init__(self):
        self.dense = MultilingualE5Large()  # vector search via pgvector
        self.sparse = BM25Index()           # keyword search
        self.reranker = CrossEncoder('cross-encoder/ms-marco-MiniLM-L-6-v2')
    async def retrieve(self, query: FarmStateQuery, top_k: int = 5):
        # Parallel dense + sparse retrieval
        dense_hits = await self.dense.search(query.to_text(), k=top_k*3,
            filters={'crop': query.crop, 'state': query.state,
                     'status': 'PUBLISHED', 'valid_until__gt': date.today()})
        sparse_hits = self.sparse.search(query.to_keywords(), k=top_k*3)
        # Merge by reciprocal rank fusion
        merged = reciprocal_rank_fusion([dense_hits, sparse_hits])
        # Cross-encoder reranking
        reranked = self.reranker.rerank(query.to_text(), merged)
        # Source authority boost: Tier 1 documents score * 1.3
        boosted = apply_authority_boost(reranked)
        return boosted[:top_k]
