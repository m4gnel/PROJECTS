2.1 GDD Crop Stage Calculation Tests
# tests/unit/test_crop_stage.py
import pytest
from datetime import date
from backend.agents.crop_stage import CropStageAgent, compute_gdd
class TestGDDCalculation:
    def test_rice_gdd_base_temperature(self):
        # Rice base temp = 10°C
        # Tmax=30, Tmin=22 → GDD = ((30+22)/2) - 10 = 16
        gdd = compute_gdd(tmax=30.0, tmin=22.0, crop='rice')
        assert gdd == pytest.approx(16.0, abs=0.1)
    def test_gdd_below_base_temp_returns_zero(self):
        # If mean temp < base temp, GDD = 0, not negative
        gdd = compute_gdd(tmax=12.0, tmin=8.0, crop='rice')
        assert gdd == 0.0
    def test_maize_base_temperature_different(self):
        # Maize base temp = 10°C — same for maize
        gdd = compute_gdd(tmax=28.0, tmin=18.0, crop='maize')
        assert gdd == pytest.approx(13.0, abs=0.1)
    def test_rice_tillering_stage_at_expected_gdd(self):
        # Rice reaches tillering at ~250 GDD after germination
        agent = CropStageAgent()
        result = agent.estimate_stage(
            crop='rice', cultivar='Jyothi',
            planting_date=date(2026, 6, 1),
            cumulative_gdd=260.0
        )
        assert result.stage_code == 'tillering'
        assert result.disease_vulnerability == 'high'
    def test_stage_vulnerability_mapping_rice(self):
        agent = CropStageAgent()
        # Rice blast most dangerous at tillering and flowering
        vuln = agent.get_disease_vulnerability('rice', 'tillering', 'rice_blast')
        assert vuln == 'high'
        vuln_low = agent.get_disease_vulnerability('rice', 'germination', 'rice_blast')
        assert vuln_low == 'low'

4.3 Vision Model Evaluation Script
# scripts/evaluate_vision_model.py
import onnxruntime as ort
import numpy as np
from sklearn.metrics import f1_score, classification_report, confusion_matrix
from sklearn.calibration import calibration_curve
import json, os
from pathlib import Path
def run_evaluation(model_path: str, golden_dir: str) -> dict:
    session = ort.InferenceSession(model_path)
    y_true, y_pred, y_prob = [], [], []
    for crop_dir in Path(golden_dir).iterdir():
        for disease_dir in crop_dir.iterdir():
            if disease_dir.name == 'ood':
                continue  # Evaluate OOD separately
            for img_file in disease_dir.glob('*.jpg'):
                label_file = img_file.with_suffix('.json')
                if not label_file.exists():
                    continue
                label = json.loads(label_file.read_text())['label']
                img = preprocess_image(img_file)
                logits = session.run(['logits'], {'image': img})[0]
                probs = softmax(logits[0])
                pred_idx = np.argmax(probs)
                y_true.append(LABEL_TO_IDX[label])
                y_pred.append(pred_idx)
                y_prob.append(probs[pred_idx])
    macro_f1 = f1_score(y_true, y_pred, average='macro')
    ece = compute_ece(y_true, y_pred, y_prob)
    results = {
        'macro_f1': macro_f1,
        'ece': ece,
        'classification_report': classification_report(y_true, y_pred, target_names=ALL_LABELS),
        'PASS': macro_f1 >= 0.82 and ece <= 0.12
    }
    if not results['PASS']:
        print(f'FAIL: macro_F1={macro_f1:.3f} (need >=0.82), ECE={ece:.3f} (need <=0.12)')
        exit(1)  # Fail CI if model does not meet bar
    return results
def compute_ece(y_true, y_pred, y_prob, n_bins=10):
    # Expected Calibration Error — measures confidence reliability
    from sklearn.calibration import calibration_curve
    fraction_pos, mean_pred = calibration_curve(
        [1 if t==p else 0 for t,p in zip(y_true,y_pred)], y_prob, n_bins=n_bins)
    return np.mean(np.abs(fraction_pos - mean_pred))
