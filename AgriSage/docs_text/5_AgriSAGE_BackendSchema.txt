3.2 Graph Wiring
def build_advisory_graph() -> StateGraph:
    graph = StateGraph(AgriSAGEState)
    # Add all nodes
    graph.add_node('intake', intake_agent.run)
    graph.add_node('vision', vision_agent.run)
    graph.add_node('weather', weather_agent.run)
    graph.add_node('crop_stage', crop_stage_agent.run)
    graph.add_node('knowledge', knowledge_agent.run)
    graph.add_node('critic', critic_agent.run)
    graph.add_node('scheme', scheme_agent.run)
    graph.add_node('simulation', simulation_agent.run)
    graph.add_node('decision', decision_agent.run)
    graph.add_node('explanation', explanation_agent.run)
    # Entry point
    graph.set_entry_point('intake')
    # Intake → parallel execution (vision + weather + stage + knowledge)
    graph.add_edge('intake', 'vision')
    graph.add_edge('intake', 'weather')
    graph.add_edge('intake', 'crop_stage')
    graph.add_edge('intake', 'knowledge')
    # All 4 parallel → critic (barrier node)
    graph.add_edge('vision', 'critic')
    graph.add_edge('weather', 'critic')
    graph.add_edge('crop_stage', 'critic')
    graph.add_edge('knowledge', 'critic')
    # Critic → scheme (always) + simulation (conditional)
    graph.add_edge('critic', 'scheme')
    graph.add_conditional_edges(
        'critic',
        lambda s: 'simulation' if s['run_simulation'] else 'decision',
        {'simulation': 'simulation', 'decision': 'decision'}
    )
    # Simulation → decision
    graph.add_edge('simulation', 'decision')
    graph.add_edge('scheme', 'decision')
    graph.add_edge('decision', 'explanation')
    graph.add_edge('explanation', END)
    return graph.compile()
advisory_graph = build_advisory_graph()

6.1 Chemical Safety Check
class ChemicalSafetyChecker:
    """
    All chemical recommendations must pass these checks.
    A BLOCK result means the advisory is held — not shown to farmer.
    """
    async def check_product_exists(self, product_name: str) -> bool:
        """Product name must exist in CIB&RC database."""
        result = await db.fetch_one(
            'SELECT id FROM chemical_labels WHERE product_name ILIKE :name',
            {'name': product_name}
        )
        return result is not None
    async def check_crop_registration(self, product_name: str, crop: str) -> bool:
        """Product must be registered for use on stated crop."""
        result = await db.fetch_one(
            'SELECT id FROM chemical_labels WHERE product_name ILIKE :name AND :crop = ANY(registered_crops)',
            {'name': product_name, 'crop': crop}
        )
        return result is not None
    async def check_dosage(self, product_name: str, stated_dose: float, unit: str, crop: str) -> bool:
        """Stated dosage must be within label-approved range."""
        label = await self.get_label(product_name, crop)
        if label is None:
            return False
        return label.min_dose <= stated_dose <= label.max_dose
    async def run_all_checks(self, recommendation: dict) -> tuple[bool, List[str]]:
        """Run all safety checks. Returns (safe, list_of_issues)."""
        issues = []
        product = recommendation.get('product_name')
        if not product:
            return True, []  # No chemical — skip
        if not await self.check_product_exists(product):
            issues.append(f'BLOCK: Product {product} not found in CIB&RC database')
        if not await self.check_crop_registration(product, recommendation['crop']):
            issues.append(f'BLOCK: {product} not registered for {recommendation["crop"]}')
        if recommendation.get('dosage'):
            if not await self.check_dosage(product, recommendation['dosage'], recommendation['unit'], recommendation['crop']):
                issues.append(f'BLOCK: Dosage outside label-approved range')
        return len(issues) == 0, issues
