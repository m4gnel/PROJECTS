# Token payload — minimum claims only
def create_access_token(user_id: str, subscription_tier: str) -> str:
    payload = {
        'sub': user_id,           # Subject — user UUID only
        'tier': subscription_tier, # For feature gating
        'exp': datetime.utcnow() + timedelta(minutes=15),
        'iat': datetime.utcnow(),
        'jti': str(uuid4()),      # JWT ID — enables token blacklisting if needed
    }
    return jwt.encode(payload, settings.SECRET_KEY, algorithm=ALGORITHM)
# Validation — backend/dependencies.py
async def require_auth(token: str = Depends(oauth2_scheme), db=Depends(get_db)):
    try:
        payload = jwt.decode(token, settings.SECRET_KEY, algorithms=[ALGORITHM])
        user_id = payload.get('sub')
        if not user_id:
            raise HTTPException(401, 'Invalid token')
        # Set PostgreSQL session variable for RLS
        await db.execute(f"SET LOCAL app.current_user_id = '{user_id}'")
        return payload
    except JWTError:
        raise HTTPException(401, 'Invalid or expired token')

2.3 Refresh Token Rotation
# On every token refresh — backend/api/v1/auth.py
async def refresh_token(refresh_token: str, db=Depends(get_db)):
    # 1. Hash the incoming token
    token_hash = hashlib.sha256(refresh_token.encode()).hexdigest()
    # 2. Find in DB — must exist, not revoked, not expired
    record = await db.fetch_one(
        'SELECT * FROM refresh_tokens WHERE token_hash=:h AND NOT revoked AND expires_at > NOW()',
        {'h': token_hash}
    )
    if not record:
        raise HTTPException(401, 'Invalid refresh token')
    # 3. Revoke old token BEFORE issuing new one
    await db.execute('UPDATE refresh_tokens SET revoked=true WHERE id=:id', {'id': record['id']})
    # 4. Issue new access + refresh token pair
    new_access = create_access_token(record['user_id'])
    new_refresh = create_refresh_token(record['user_id'])
    return {'access_token': new_access, 'refresh_token': new_refresh}

4.1 Complete Image Upload Security Pipeline
# backend/services/image_security.py
import magic          # python-magic — MIME from file content, not extension
import subprocess     # ExifTool for metadata strip
from PIL import Image
import io
ALLOWED_MIME_TYPES = {'image/jpeg', 'image/png', 'image/webp'}
MAX_SIZE_BYTES = 5 * 1024 * 1024  # 5MB
MAX_DIMENSION_PX = 1200
async def process_uploaded_image(file_bytes: bytes, original_filename: str) -> bytes:
    # 1. Size check BEFORE reading into memory fully
    if len(file_bytes) > MAX_SIZE_BYTES:
        raise ValueError(f'Image too large: {len(file_bytes)} bytes. Max: {MAX_SIZE_BYTES}')
    # 2. MIME check from content — NOT from filename/extension
    mime = magic.from_buffer(file_bytes[:2048], mime=True)
    if mime not in ALLOWED_MIME_TYPES:
        raise ValueError(f'File type not allowed: {mime}')
    # 3. Open with Pillow — verifies it is a real image
    try:
        img = Image.open(io.BytesIO(file_bytes))
        img.verify()  # Detect truncated/corrupt images
        img = Image.open(io.BytesIO(file_bytes))  # Reopen after verify
    except Exception as e:
        raise ValueError(f'Invalid image: {e}')
    # 4. Strip ALL EXIF metadata — protects farmer location privacy
    # Create new image without metadata
    clean = Image.new(img.mode, img.size)
    clean.putdata(list(img.getdata()))
    # 5. Resize to max dimension while preserving aspect ratio
    if max(clean.size) > MAX_DIMENSION_PX:
        clean.thumbnail((MAX_DIMENSION_PX, MAX_DIMENSION_PX), Image.LANCZOS)
    # 6. Re-encode as JPEG (normalizes format, removes any embedded scripts)
    output = io.BytesIO()
    clean.save(output, format='JPEG', quality=85, optimize=True)
    return output.getvalue()

name: Security Gates
on: [push, pull_request]
jobs:
  security:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Check for hardcoded secrets

          # Fail if any pattern like API_KEY="actual_value" found

             -E '(SECRET_KEY|API_KEY|PASSWORD|TOKEN)\s*=\s*"[a-zA-Z0-9+/]{20,}"' .; then
            echo 'FAIL: Hardcoded secret detected'
            exit 1
          fi
      - name: Dependency vulnerability scan
        run: |
          pip install pip-audit
          pip-audit -r requirements.txt --fail-on CRITICAL
